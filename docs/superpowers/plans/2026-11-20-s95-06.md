- [ ] 增加差异纯函数并用原输入和服务端确认的 expected 调用，不能使用尚未保存的本地目标判分。

```javascript
export function spellingDifference(typed, expected) {
  const actual = Array.from(typed || '');
  const target = Array.from(expected || '');
  let start = 0;
  while (start < actual.length && start < target.length
    && actual[start].toLowerCase() === target[start].toLowerCase()) start += 1;
  if (start === actual.length && start === target.length) return '';
  let endActual = actual.length;
  let endTarget = target.length;
  while (endActual > start && endTarget > start
    && actual[endActual - 1].toLowerCase() === target[endTarget - 1].toLowerCase()) {
    endActual -= 1;
    endTarget -= 1;
  }
  const written = actual.slice(start, endActual).join('');
  const required = target.slice(start, endTarget).join('');
  if (!written) return `第 ${start + 1} 处缺少：${required}`;
  if (!required) return `第 ${start + 1} 处多写：${written}`;
  const unit = Array.from(written).length === 1 && Array.from(required).length === 1
    ? '个字母' : '处';
  return `第 ${start + 1} ${unit}：${written} → ${required}`;
}
```
